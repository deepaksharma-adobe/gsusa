10 — Token Quick Reference

Copy-paste reference for the most-used tokens.

/* Colors */
--color-gs-green:    #00b451   /* Primary CTA */
--color-gs-forest:   #005640   /* Hover */
--color-gs-poppy:    #ee3124   /* Error / sale */
--color-gs-gold:     #f7be00   /* Rating */

/* Spacing (GS scale) */
--spacing-xxsmall:   4px        /* GS: Small */
--spacing-small:     16px       /* GS: Large */
--spacing-medium:    24px       /* GS: XL */
--spacing-large:     64px       /* GS: 4XL */

/* Radius / Shadow */
--radius-sm:   6px      --shadow-sm:  card default
--radius-md:   8px      --shadow-md:  card hover
--radius-full: 9999px   --shadow-lg:  modal / drawer

/* Motion */
--transition-fast:   150ms ease
--transition-base:   250ms ease
--transition-slow:   400ms ease

/* Use in blocks */
.my-block {
  color: var(--color-gs-green);
  padding: var(--spacing-medium);
  border-radius: var(--radius-md);
  box-shadow: var(--shadow-sm);
  transition: color var(--transition-base);
}
